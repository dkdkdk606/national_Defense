9. 레이터 차트
- 다변량 데이터를 2차원에 표현하여 관측값들을 비교할 때 사용
- 레이더 차트(radar chart)는 방사형 차트라고도 불리며 여러 변수의 값을 비교하는 데 유용한 시각화 도구
- 변수의 수에 따라 원을 같은 간격으로 나누고, 중심으로부터 일정 간격으로 동심원을 형성하며 척도를 측정하는 축을 만듬
- 각 변수의 정량화된 점수를 축 위에 표시하고, 해당 점들을 선으로 연결하여 다각형 모양을 형성함
- 레이더 차트의 대표적인 활용 분야 중 하나는 여러 제품에 대한 품질 평가

# plotly 설치
# !pip install plotly
# !pip install kaleido

import matplotlib.image as mpimg
import plotly.graph_objects as go
import pandas as pd
import matplotlib.pyplot as plt

def radar(df, fills, min_max, title=''):
    fig = go.Figure()
    categories = df.columns.to_list()
    categories.append(categories[0])
    i = 0
    while (i < len(df)):
        scores = df.iloc[i, :].to_list()
        scores.append(scores[0])
        fig.add_trace(go.Scatterpolar(
            r=scores,  # 축의 값
            theta=categories,  # 축의 레이블
            fill=fills[i],  # 다각형 채우기 색
            name=df.index[i]  # 다각형 레이블
        ))
        i += 1
        
        fig.update_layout(
            polar_radialaxis_visible=True,
            polar_radialaxis_range=min_max,  # 축의 값 범위
            showlegend=True,
            margin_t=50,  # 상단 여백
            margin_l=100,  # 좌측 여백
            margin_r=100,  # 우측 여백
            margin_b=25,  # 하단 여백
            width=700,  # 그래프의 폭(pixel)
            height=700,  # 그래프의 높이(pixel)
            title_text=title,  # 그래프 제목
            title_font_size=30,  # 제목 폰트 사이즈
            font_size=20  # 폰트 사이즈
            )
        
    # 그래프 저장 & display
    plt.axis('off')
    fig.write_image('rader.png')
    plt.imshow(mpimg.imread('rader.png'))
    plt.show()

# 예시데이터
df = pd.DataFrame({
    'Kor': [72, 70, 90, 60, 66],
    'Eng': [84, 85, 95, 70, 85],
    'Math': [71, 40, 88, 80, 75],
    'Sci': [83, 80, 91, 90, 70],
    'Phy': [60, 60, 60, 70, 50]
})
df.index = ['AVG', 'John', 'Tom', 'Smith', 'Grace']

fills = [None, 'toself']
radar(df=df.iloc[[0, 3], :],
      fills=fills,
      min_max=[0, 100],
      title='Scores of Smith'
      )

fills = ['toself', 'toself', 'toself', 'toself']
radar(df=df.iloc[1:, :],
      fills=fills,
      min_max=[0, 100],
      title='Scores of Student'
      )

10. 트리맵(tree map)
- 계층 구조를 포함하는 데이터를 2차원으로 시각화하는 도구
- 모자이크 플롯은 데이터의 그룹을 하나의 타일로 표현하는 반면, 트리맵에서는 각 관측값이 하나의 타일로 표현
- 트리맵에서는 타일의 색과 크기를 활용하여 서로 다른 정보를 동시에 표현할 수 있으며, 타일들은 계층 구조에 따라 배치됨
- Plotly 라이브러리에서 트리맵을 지원
- 트리맵 작성과 화면 출력을 위해 plotly와 webbrowser 라이브러리(패키지)를 설치해야 함

import plotly.express as px
import webbrowser
import matplotlib.pyplot as plt
import pandas as pd

df = pd.read_csv('data/GNI2014.csv')

fig = px.treemap(data_frame=df,
                 path=['continent', 'iso3'],  # 데이터의 계층 구조
                 values='population',  # 타일 면적 기준 컬럼
                 color='GNI',  # 색 온도 기준 컬럼
                 color_continuous_scale='Bluyl'  # 컬러 팔레트
                 )

plt.axis('off')  # 축 눈금 제거
fig.update_layout(margin_t=50, margin_l=25,  # 여백 설정
                  margin_r=25, margin_b=25,
                  width=800,  # 그래프의 폭(pixel)
                  height=600,  # 그래프의 높이(pixel)
                  title_text='GNI 2014',  # 그래프 제목
                  title_font_size=20  # 제목 폰트 크기
                  )

# 그래프 저장 & 화면에 표시하기
fig.write_html('treemap.html')  # html 파일로 결과 저장
webbrowser.open('treemap.html')  # 웹 브라우저에서 그래프 확인
